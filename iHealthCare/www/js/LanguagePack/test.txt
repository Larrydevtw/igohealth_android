        translateByMSFTApi:function(fromLocale,lToLocale,_lTexts,successFn){
            //陣列最多可以有 25 個項目。
            //要求中包含的完整文字不能超過 5,000 個字元，包括空格。
            if(!fromLocale){fromLocale= "zh-Hant";};
            if(!lToLocale){lToLocale.push("zh-Hans");};
            var lTexts=[{
                "Text":_lTexts[0]
            }];

            var apiUrl = "https://api.cognitive.microsofttranslator.com/translate?api-version=3.0&from=" + fromLocale+ "&to=" + lToLocale[0];
            if(lTexts.length>1){
                apiUrl+="&to="+lToLocale[1];
            }

            var fromTexts = JSON.stringify(lTexts);
            $.ajax({
                url: apiUrl,
                headers: {
                    'Ocp-Apim-Subscription-Key': '31ba41584d3f402b98e0499dfc14a20a',
                    'Content-Type': 'application/json'
                },
                method: 'POST',
                dataType: 'json',
                data: fromTexts,
                async: false,
                success: function (data) {
                    var dataLength=data.length;
                    //console.log(data);
                    //console.log('succes: ' + JSON.stringify(data));
                    var result = "";
                    //console.log(dataLength);
                    $.each(data, function (i) {
                        //console.log(data[i].translations[0].text);
                        result += data[i].translations[0].text;
                        if (i === dataLength - 1) {
                            //console.log(result);
                            //console.log(result.length);
                            result=result.replace(/、/g,",").replace(/「/g,"\"").replace(/」/g,"\"");
                            if(successFn){
                                result=result.split("^");
                                successFn(result);
                            }
                        }
                    });
                },
                error:function(err){
                    console.log(err);
                }               
                
            });
        },

        getTranslationByLanguage: function (languageProperty) {
            //alert("getTranslationByLanguage");
            if (!languageProperty)
                throw new ReferenceError('languageProperty not defined');

                this.constants.language.default;
